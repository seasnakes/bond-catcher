REF_C := REF(C, 1);
C_DIFF := C - REF_C;
CHG_D := C_DIFF / REF_C * 100;
AMP_D := (H - L) / REF_C * 100;
BODY_D := (O - C) / REF_C * 100;

VM3 := MA(VOL, 3);
VM5 := MA(VOL, 5);
VM10 := MA(VOL, 10);

ST_T1 := EMA(EMA(C,10),10);
LT_T1 := (MA(C,14)+MA(C,28)+MA(C,57)+MA(C,114))/4;

HL9 := HHV(H, 9) - LLV(L, 9) + 0.001;
RSV := (C - LLV(L, 9)) / HL9 * 100;
K := SMA(RSV,3,1);
D := SMA(K,3,1);
J := 3*K-2*D;

HL4 := HHV(H, 4) - LLV(L, 4) + 0.001;
HM1 := (HHV(H, 4) - C) / HL4 * 100 - 90;
HM3 := (C - LLV(L, 4)) / HL4 * 100;
HM2 := SMA(HM1, 4, 1);
HM5 := SMA(SMA(HM3, 6, 1), 6, 1);
HM6 := HM5 - HM2;
BRICK := IF(HM6 > 4, HM6 - 4, 0);
BRICK_REF := REF(BRICK, 1);
RED_1ST := (REF(BRICK, 2) >= BRICK_REF) AND (BRICK > BRICK_REF); 

C_DAYS := BARSLAST(CROSS(ST_T1, LT_T1));
HONEYMOON := C_DAYS >= 0 AND C_DAYS <= 30 AND ST_T1 > LT_T1; 
BREAKAWAY := ST_T1 > LT_T1 * 1.03; 
WAIVER := HONEYMOON OR BREAKAWAY; 

LT_DIR := IF(BARSCOUNT(C) > 114, IF(LT_T1 > REF(LT_T1, 1), 1, -1), 1);
LT_FILTER := COUNT(LT_DIR != REF(LT_DIR, 1), 30) <= 2 OR WAIVER;

HIGH_POS := (HHV(H, 20) - LLV(L, 20)) / LLV(L, 20) * 100 > 15; 
VOL_BIG := VOL > VM5*1.3 OR VOL > VM10*1.5;
BAD_DUMP := (BODY_D > 6.0 OR CHG_D < -5.5) AND VOL_BIG AND HIGH_POS;
DUMP_DAY := BARSLAST(BAD_DUMP);

COOL_OFF := IF(COUNT(BAD_DUMP, 10) >= 2, 10, 5); 
IN_REV := DUMP_DAY >= COOL_OFF AND DUMP_DAY <= 15;
SHAPE_OK := AMP_D <= 10.0 AND CHG_D >= -4.0 AND CHG_D <= 4.0;
M5 := MA(C,5);
CG_OK := (C > M5) OR (M5 >= REF(M5,1)) OR (ABS(C-M5)/M5*100 < 1.5); 
PASS_REV := IF(IN_REV, SHAPE_OK AND CG_OK, 1);
SAFE_MODE := (DUMP_DAY >= COOL_OFF) AND PASS_REV;

SW_H := HHV(H, 30); 
SW_L := LLV(L, 30);
SW_UP := (SW_H - SW_L) / SW_L * 100;
CUR_DD := (SW_H - L) / SW_H * 100;
FLAT_Z := (SW_UP > 40 AND CUR_DD / (SW_UP + 0.001) < 0.35);
HLTH_SPC := FLAT_Z OR (SW_UP <= 40 AND CUR_DD < 20);
V_SHRINK := VM3 < VM10; 

ENV_OK := ST_T1 > LT_T1 AND C >= LT_T1 * 0.95 AND CHG_D > -3.0 AND SAFE_MODE AND V_SHRINK AND HLTH_SPC AND LT_FILTER;
PHYS_OK := (VOL <= LLV(VOL, 10) * 1.1 AND ABS(BODY_D) < 3.7) OR V_SHRINK;
RSI6 := SMA(MAX(C_DIFF, 0), 6, 1) / SMA(ABS(C_DIFF), 6, 1) * 100;
J_HIS := IF(FLAT_Z, LLV(J, 8) < 30, LLV(J, 8) < 22);
J_CUR := IF(FLAT_Z, J < 35, J < 30);

B1_Y := ENV_OK AND RED_1ST AND J_CUR AND J_HIS AND RSI6 > REF(RSI6, 1) AND CHG_D <= 4.0 AND C > O;
B1_W := ENV_OK AND J < 16 AND PHYS_OK AND CHG_D <= 4.0 AND NOT(B1_Y);

JC_P2 := VALUEWHEN(CROSS(ST_T1, LT_T1), C);
MAX_P2 := HHV(H, C_DAYS + 1);
DOUBLE_V := (VOL >= REF(VOL,1)*1.9 OR VOL >= VM5*2) AND C > O;
BL_V2 := VALUEWHEN(DOUBLE_V, VOL); 
YIN_L2 := C < O OR C < REF_C;

ACCUM_BASE := ((C - JC_P2) / JC_P2 * 100 < 40) 
      AND ((MAX_P2 - C) / MAX_P2 * 100 < 22) AND (C > LT_T1 * 0.95 AND C < LT_T1 * 1.15) 
      AND (COUNT(DOUBLE_V, 30) >= 2) 
      AND (COUNT(YIN_L2 AND VOL >= BL_V2, 10) = 0) AND (COUNT(YIN_L2 AND VOL > VM10 * 1.2, 10) < 2) 
      AND V_SHRINK AND J < 16 AND SAFE_MODE;

ACCUM_BLOCK := COUNT(REF(ACCUM_BASE, 1), 15) > 0 AND J > REF(J, 1);
B1_Q := ACCUM_BASE AND NOT(B1_Y) AND NOT(B1_W) AND NOT(ACCUM_BLOCK);
TOP_RES := RED_1ST AND (B1_W OR B1_Y);

B1_Y OR B1_W OR B1_Q OR TOP_RES;

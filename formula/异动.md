N:= 30;
RSV:= (CLOSE - LLV(LOW, 9)) / (HHV(HIGH, 9) - LLV(LOW, 9)) * 100;
K:= SMA(RSV, 3, 1);
D:= SMA(K, 3, 1);
J:= 3 * K - 2 * D;
{ 2. 计算MACD的DIF值 }
SHORT := 12;
LONG := 26;
DIF := EMA(CLOSE, SHORT) - EMA(CLOSE, LONG);
VOL30 := VOL;
VH1 := HHV(VOL30, N);
VH2 := HHV(IF(VOL30 < VH1, VOL30, 0), N);
VH3 := HHV(IF(VOL30 < VH2, VOL30, 0), N);
AVGHIGH := (VH1 + VH2 + VH3) / 3;
VL1 := LLV(VOL30, N);
VL2 := LLV(IF(VOL30 > VL1, VOL30, 9999999), N);
VL3 := LLV(IF(VOL30 > VL2, VOL30, 9999999), N);
AVGLOW := (VL1 + VL2 + VL3) / 3;
COND1 := J < 13;
COND2 := DIF > 0;
COND3 := AVGHIGH > 3.5 * AVGLOW;
SELECT: COND1 AND COND2 AND COND3;
